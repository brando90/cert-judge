  % CS197 moves 1-2: motivation folded into the single italic hook line
  \begin{block}{Validating LLM judges one human study at a time does not scale}

    {\centering\itshape Agents write Lean~4 code and its specs faster than humans can inspect --- can one human study, via automatic falsifiable checks, be amortized across every future judge?\par}

  \end{block}

  % CS197 move 3: the flip — certify with falsifiable behavioral checks instead of re-labeling
  \begin{alertblock}{Key idea: certify the judge --- don't re-label it}

    \begin{itemize}
      \item \textbf{CertJudge}: rate a small panel \textbf{once} $\rightarrow$ \textbf{automatic falsifiable property checks} + \tivar{} certify every new judge --- no new labels.
      \item Human panel: $n{=}75$ candidate--gold Lean pairs, single annotator, two rubric passes; 8 judge configs per family panel.
    \end{itemize}

  \end{alertblock}

  % CS197 move 6: takeaways — vision, not self-congratulation; caveat is part of the claim
  \vspace{0.6cm}
  \begin{alertblock}{Takeaways}

    \begin{itemize}
      \setlength{\itemsep}{0.8ex}
      \item \textbf{Amortized oversight:} one $n{=}75$ panel + automatic falsifiable checks screen every new judge --- no re-labeling.
      \item \textbf{Measure trust, don't assume it:} behavioral properties expose failures human-agreement scores miss.
      \item \textbf{Scope:} \emph{development-set evidence, not universal reliability} --- locked held-out test (5 raters, 343 labels, 78 pairs) is next.
    \end{itemize}

  \end{alertblock}
